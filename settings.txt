# ================= fastollama settings =================
# Lines starting with # are comments. Format: key = value
# Restart fastollama after changing values.

# --- Model ---
# Model name inside models/ folder (without .gguf) or full path
# quant guide: q8_0 = best quality/slowest, q6_k, q5_k_m, q4_k_m = fast, iq3/q3 = max speed
# auto = walk the quant ladder, pick the biggest that fits the VRAM budget
model = Qwen3.8-27B-UD-IQ2_XXS

# MEASURED on this PC (full 262144 context, VRAM <= 15 GB, MTP on, full_gpu = 1):
#   Qwen3.8-27B-UD-IQ2_S      80-86 t/s gen / 1000 t/s pp  <-- DEFAULT: whole 27B on GPU!
#   Qwen3.8-27B-UD-IQ2_XXS    86 t/s gen / 179 t/s pp  (max speed, more quality risk)
#   Qwen3.8-27B-UD-IQ3_XXS     4.5 t/s gen /  33 t/s pp  (CPU-offload split, quality+)
#   Qwen3.8-27B-UD-Q4_K_XL     6.9 t/s gen /  27 t/s pp  (best 27B quality)
#   Qwen3-30B-A3B ...Q4_K_XL  30.5 t/s gen / 105 t/s pp  (MoE, best quality-per-tok)
#     ^ same 30B at context = 40960: 40.3 t/s gen / 154 t/s pp
# Speed rule of thumb on this box: ANY CPU-resident tensor (even FFN) drops the
# dense 27B to single digits; a fully-GPU quant is a different speed class.
# Switch to the 30B: set model = Qwen3-30B-A3B-Instruct-2507-UD-Q4_K_XL
# (its MTP sidecar does not match its arch; fastollama auto-skips MTP for it).
# Old small models still work: Qwen3-8B-IQ4_XS (~106 t/s), Qwen3-4B-Q3_K_S (~167 t/s via API).

# --- Backend ---
# vulkan = best speed on RX 9070 XT (recommended), hip = ROCm/HIP, cpu = no GPU
backend = vulkan

# --- VRAM safety governor ---
# The engine is never allowed to use more than vram_limit_gb of VRAM, and at
# least vram_free_min_gb always stays free for your desktop/games.
#
# full_gpu = 1 (RECOMMENDED): put the WHOLE model + KV cache on the GPU.
# If the quant is too big to fit, the governor falls back to the smart split
# automatically - you never have to tune anything by hand.
#
# auto_vram = 1: fastollama reads the GGUF, computes the exact KV-cache and
# compute-buffer sizes and picks the layer split for you. Whole layers spill
# to your 32 GB RAM automatically instead of OOM-crashing GNOME.
# Set auto_vram = 0 to use the fixed gpu_layers value below instead.
vram_limit_gb = 15.0
vram_free_min_gb = 0.6
# Last-resort watchdog: if VRAM use ever crosses vram_emergency_pct of total
# (driver peak, leak, second app grabbing memory) the server is stopped cleanly
# instead of freezing GNOME. vram_guard = 0 disables.
vram_guard = 1
vram_emergency_pct = 0.95
auto_vram = 0
# ot_mode: layers = classic whole-layer split (ngl), smart = keep ALL attention/KV
# layers on GPU and offload only the heavy FFN/SSM tensors of the remaining layers
# via -ot regex. Best for hybrid archs (qwen35/qwen3-next) and long context.
ot_mode = smart
# gpu_layers is only used when auto_vram = 0
gpu_layers = 999

# --- Context / memory ---
# context = 262144 -> Qwen3.8-27B native max (no yarn needed)
# context = 40960  -> Qwen3-8B native max
context = 262144

# kv_cache: f16 = full quality / most VRAM, q8_0 = ~2x smaller (~0.1% loss),
# q4_0 = 4x smaller (use for 256k context; needed so most layers stay on GPU)
kv_cache = q4_0

# mlock = 1 pins model in RAM so it never swaps. no_mmap = 1 loads fully to RAM first.
# no_mmap+mlock: -ot CPU expert tensors get a contiguous pinned RAM copy -> no page
# faults during generation (engine log explicitly recommended load-mode none for -ot).
mlock = 1
no_mmap = 1

# defrag threshold: compact fragmented KV cache when free slots drop below (0 = off)
defrag = 0.1
# kv_unified = 1: one big KV pool instead of per-layer slabs (less fragmentation
# at huge context; engine aborts safely if unsupported)
kv_unified = 0
# cache_reuse = N: after context shift, reuse N tokens of cached prompt instead
# of re-processing them (huge for repeated prefixes)
cache_reuse = 256
# no_context_shift = 1: never silently rewrite the prompt when the context fills
# up (error out instead). Recommended when you manage truncation yourself.
no_context_shift = 0

# --- Speed ---
# CPU threads: MEASURED on 9700X (Zen5, SMT) for Qwen3-Next-80B MoE @ 30 CPU expert
# layers: 4T=32, 6T=34, 8T=34, 12T=32, 16T=29 t/s. Expert math is RAM-bound:
# 6 threads wins and leaves 10 cores for the desktop.
threads = 6
threads_batch = 6
batch = 2048
ubatch = 1024

# flash attention: 1 = ON (always keep on, it is pure speed+memory win here)
flash_attn = 1

# --- MTP / speculative decoding ---
# spec_type = draft-mtp: MTP sidecar drafts tokens, the big model verifies in one pass.
#   Both sidecars shipped for Flash-Next currently fail our engine's loader
#   (missing token_embd / output_hc_norm tensors) — so it is OFF for Flash-Next.
#   For Qwen3.8-27B use: spec_type = draft-mtp + mtp_model = mtp-Qwen3.8-27B-Q4_0
# spec_type = draft: classic small-draft-model speculative decoding.
# spec_type = off (or empty + speculative = 0) disables it.
# spec_type = draft: draft-model speculation. Qwen3-0.6B-Q8 shares the 80B's exact
# tokenizer (required). Needs the PATCHED llama.cpp (FASTOLLAMA_DRAFT_CTX env)
# that shrinks draft ctx: draft_ctx = 8192 -> drafter KV ~270 MB instead of ~7.5 GB.
# MEASURED MATH: with 37 CPU expert layers verification scales ~linearly with the
# batch, so draft speculation LOSES to plain generation on this 16 GB card
# (~23 vs ~34 t/s). It wins only with ALL experts on GPU (24 GB card).
# Keep spec_type = ngram for the free +0.6 t/s; see docs/80ts-plan.md.
spec_type = ngram
draft_model = Qwen3-0.6B-Q8_0
draft_ctx = 8192
gpu_layers_draft = 999
mtp_model = mtp-Qwen3.8-27B-Q4_0
draft_max = 6
draft_min = 1
draft_p_min = 0.6
# legacy switch for spec_type = draft (ignored when spec_type = draft-mtp)
speculative = 0
# draft_model = Qwen3-0.6B-IQ4_XS
draft_p_min = 0.75

# full_gpu = 1: force the WHOLE model + KV on the GPU (fastest; needs quant that fits)

# calibrated: measured engine VRAM floor with experts offloaded on this machine.
# Qwen3-Next-80B IQ2_XXS @ 262K: engine fixed (attn+ssm+KV+compute+output) = 4.25 GiB
